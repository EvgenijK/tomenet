Metric script used (temporary feedback loop, not a new suite):
from PIL import Image, ImageFilter
import sys
base=Image.open('/tmp/sv-blur/before.bmp').convert('RGB').crop((0,0,1440,60))
actual=Image.open(sys.argv[1]).convert('RGB').crop((0,0,1440,60))
reference=base.filter(ImageFilter.GaussianBlur(2.08))
def hf(im):
 a=list(im.convert("L").getdata()); w,h=im.size
 return sum((a[y*w+x]-a[y*w+x-1])**2+(a[y*w+x]-a[(y-1)*w+x])**2 for y in range(1,h) for x in range(1,w))/((w-1)*(h-1))
ratio=hf(actual)/hf(base); target=hf(reference)/hf(base)
print(f'native HF/baseline={ratio:.4f}; CSS contrast(.78) blur(2px) transformed sigma=2.08 expected HF ratio~{target*.78**2:.4f}')
assert ratio < .15, 'Background remains sharp: CSS-like blur missing'
